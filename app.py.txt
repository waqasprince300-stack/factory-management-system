MAX_BYTES = 20 * 1024 * 1024
MAX_PAGES = 30
Image.MAX_IMAGE_PIXELS = 25000000


def ocr(image):
    image = ImageOps.autocontrast(ImageOps.grayscale(ImageOps.exif_transpose(image)))
    return pytesseract.image_to_string(image, config="--psm 6", timeout=45)


def extract(upload):
    raw = upload.getvalue()
    if len(raw) > MAX_BYTES:
        raise ValueError("File exceeds the 20 MB limit.")
    texts = []
    if upload.name.lower().endswith('.pdf'):
        if not raw.startswith(b'%PDF-'):
            raise ValueError("The file is not a valid PDF.")
        with pdfplumber.open(io.BytesIO(raw)) as pdf:
            if len(pdf.pages) > MAX_PAGES:
                raise ValueError("PDF exceeds 30 pages. Split it into smaller files.")
            for page in pdf.pages:
                if page.width * page.height > 2000000:
                    raise ValueError("PDF page dimensions exceed the safe rendering limit.")
                text = page.extract_text(layout=False) or ''
                if len(re.sub(r'\s', '', text)) < 35:
                    text = ocr(page.to_image(resolution=150).original)
                texts.append(text)
    else:
        with warnings.catch_warnings():
            warnings.simplefilter('error', Image.DecompressionBombWarning)
            with Image.open(io.BytesIO(raw)) as image:
                if image.format not in ('PNG', 'JPEG'):
                    raise ValueError("Only PNG and JPEG images are supported.")
                texts.append(ocr(image))
    result = '\n'.join(texts).strip()
    if not result:
        raise ValueError("No readable text found. Try a sharper scan.")
    return result


def parse(text):
    # Conservative label matching; missing or ambiguous fields require manual review.
    def field(pattern):
        values = re.findall(pattern, text, flags=re.I | re.M)
        parsed = [float(v.replace(',', '')) for v in values]
        return parsed[0] if len(set(parsed)) == 1 else 0.0
    stitches = field(r'^\s*(?:total\s+)?stitches\s*[:=]?\s*([\d,]+)\b')
    meters = field(r'^\s*(?:total\s+)?(?:upper\s+)?thread\s+(?:length|usage|consumption)\s*[:=]?\s*([\d,]+(?:\.\d+)?)\s*(?:m|meters|metres)\b')
    return stitches, meters
